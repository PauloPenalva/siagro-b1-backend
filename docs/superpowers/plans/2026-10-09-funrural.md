# Funrural no Documento de Entrada — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Calcular o Funrural (INSS + RAT + SENAR) por linha no Documento de Entrada a partir das alíquotas da natureza de operação, mostrar retido e líquido a pagar, montar o texto da NF-e e estornar o mesmo Funrural na devolução de compra.

**Architecture:** Uma função pura no `SiagroB1.Fiscal` calcula as três parcelas e outra monta o texto; a natureza (`USAGES`) ganha três alíquotas; a linha de entrada guarda a fotografia (base, alíquotas, valores) gravada por `PurchaseInvoicesTaxApplyService`, no mesmo ponto e com a mesma trava do ICMS. A devolução copia as alíquotas da linha de origem. O frontend mostra os campos na natureza, no diálogo fiscal da linha e nas seções "Tributos" e "Totais".

**Tech Stack:** .NET 10, EF Core (SQL Server; testes em EF InMemory), xUnit, OData v4; OpenUI5 1.141 + TypeScript, QUnit.

**Spec:** `docs/superpowers/specs/2026-10-09-funrural-design.md` (repo backend).

```csharp
namespace SiagroB1.Fiscal.Taxes;

/// <summary>Funrural da linha: a base e as três parcelas, já arredondadas.</summary>
public sealed record FunruralResult(
    decimal Base, decimal InssRate, decimal RatRate, decimal SenarRate,
    decimal InssValue, decimal RatValue, decimal SenarValue)
{
    /// <summary>Linha sem Funrural: tudo zero, inclusive a base, para "sem Funrural" ser inequívoco.</summary>
    public static readonly FunruralResult None = new(0m, 0m, 0m, 0m, 0m, 0m, 0m);

    public decimal Total => InssValue + RatValue + SenarValue;
}

/// <summary>
/// Contribuição do produtor rural retida pelo comprador (spec 2026-10-09 §3.2): INSS, RAT e SENAR sobre o total
/// geral da linha. Alíquota vazia ou zero = parcela não retida; as três zeradas = linha sem Funrural. PF, PJ e
/// decisão judicial são naturezas diferentes — aqui só chegam as alíquotas.
/// </summary>
public static class FunruralCalculator
{
    public static FunruralResult Calculate(decimal amount, decimal? inssRate, decimal? ratRate, decimal? senarRate)
    {
        var inss = inssRate ?? 0m;
        var rat = ratRate ?? 0m;
        var senar = senarRate ?? 0m;

        if (inss == 0m && rat == 0m && senar == 0m)
            return FunruralResult.None;

        return new FunruralResult(amount, inss, rat, senar, Part(amount, inss), Part(amount, rat), Part(amount, senar));
    }

    private static decimal Part(decimal amount, decimal rate) =>
        decimal.Round(amount * rate / 100m, 2, MidpointRounding.AwayFromZero);
}
```

```csharp
using System.Globalization;

namespace SiagroB1.Fiscal.Taxes;

/// <summary>O que o texto lê de cada linha: alíquotas e valores das três parcelas.</summary>
public sealed record FunruralTextLine(
    decimal InssRate, decimal RatRate, decimal SenarRate, decimal InssValue, decimal RatValue, decimal SenarValue);

/// <summary>
/// Texto do Funrural nas Informações Complementares da NF-e (spec 2026-10-09 §6), no lugar do que o operador
/// digitava à mão. Linhas com as mesmas alíquotas somam; cada combinação distinta vira um trecho.
/// </summary>
public static class FunruralText
{
    private static readonly CultureInfo PtBr = CultureInfo.GetCultureInfo("pt-BR");

    public static string? Build(IEnumerable<FunruralTextLine> lines, bool isReturn)
    {
        var sentences = lines
            .Where(l => l.InssValue + l.RatValue + l.SenarValue > 0m)
            .GroupBy(l => (l.InssRate, l.RatRate, l.SenarRate))
            .Select(g => Describe(g.Key.InssRate, g.Key.RatRate, g.Key.SenarRate,
                g.Sum(l => l.InssValue), g.Sum(l => l.RatValue), g.Sum(l => l.SenarValue)))
            .Select(text => isReturn ? $"Estorno - {text}" : text)
            .ToList();

        return sentences.Count == 0 ? null : string.Join(" | ", sentences);
    }

    private static string Describe(decimal inssRate, decimal ratRate, decimal senarRate, decimal inss, decimal rat, decimal senar)
    {
        // Decisão judicial: só o SENAR é retido, e a frase não fala em Funrural.
        if (inssRate == 0m && ratRate == 0m)
            return $"SENAR {Rate(senarRate)}% = R$ {Money(senar)}";

        var parts = new List<string>();
        if (inssRate > 0m) parts.Add($"INSS {Rate(inssRate)}% R$ {Money(inss)}");
        if (ratRate > 0m) parts.Add($"RAT {Rate(ratRate)}% R$ {Money(rat)}");
        if (senarRate > 0m) parts.Add($"SENAR {Rate(senarRate)}% R$ {Money(senar)}");

        return $"Contribuição Social denominada Funrural {Rate(inssRate + ratRate + senarRate)}% = " +
               $"R$ {Money(inss + rat + senar)} ({string.Join("; ", parts)})";
    }

    /// <summary>Duas casas no mínimo, até quatro (a alíquota é DECIMAL(7,4)).</summary>
    private static string Rate(decimal value) => value.ToString("0.00##", PtBr);

    private static string Money(decimal value) => value.ToString("N2", PtBr);
}
```

```csharp
using SiagroB1.Domain.Entities;
using SiagroB1.Fiscal.Taxes;

namespace SiagroB1.Application.Services.PurchaseInvoices;

/// <summary>
/// Escritor único da fotografia do Funrural na linha do Documento de Entrada (spec 2026-10-09 §3–§6). A linha de saída
/// não tem esses campos, por isso eles ficam fora de <c>INfeTaxedLine</c>/<c>TaxSnapshot</c>.
/// </summary>
public static class PurchaseInvoiceFunrural
{
    /// <summary>Calcula sobre o total geral da linha (a mesma base do ICMS) e grava.</summary>
    public static void Apply(PurchaseInvoiceItem item, decimal? inssRate, decimal? ratRate, decimal? senarRate) =>
        Write(item, FunruralCalculator.Calculate(item.GrandTotal, inssRate, ratRate, senarRate));

    /// <summary>Linha sem Funrural: descarta o que tenha vindo no corpo da requisição.</summary>
    public static void Clear(PurchaseInvoiceItem item) => Write(item, FunruralResult.None);

    /// <summary>Trecho do infCpl; na devolução de compra, prefixado como estorno.</summary>
    public static string? Text(PurchaseInvoice invoice) =>
        FunruralText.Build(
            invoice.Items.Select(i => new FunruralTextLine(
                i.FunruralInssRate, i.FunruralRatRate, i.FunruralSenarRate,
                i.FunruralInssValue, i.FunruralRatValue, i.FunruralSenarValue)),
            invoice.IsNfeReturn);

    private static void Write(PurchaseInvoiceItem item, FunruralResult r)
    {
        item.FunruralBase = r.Base;
        item.FunruralInssRate = r.InssRate;
        item.FunruralRatRate = r.RatRate;
        item.FunruralSenarRate = r.SenarRate;
        item.FunruralInssValue = r.InssValue;
        item.FunruralRatValue = r.RatValue;
        item.FunruralSenarValue = r.SenarValue;
    }
}
```
